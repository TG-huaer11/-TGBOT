    function showTab(id) {
      document.querySelectorAll('#tg .tab-content').forEach(el => el.classList.remove('active'));
      document.querySelectorAll('#tg .tab-btn').forEach(el => el.classList.remove('active'));
      document.getElementById(id).classList.add('active');
      const buttons = document.querySelectorAll('#tg .tab-btn');
      buttons.forEach(btn => {
        if (btn.getAttribute('onclick').includes("'" + id + "'")) {
          btn.classList.add('active');
        }
      });
    }

    function openLightbox(src) {
      const lightbox = document.getElementById('lightbox');
      const img = document.getElementById('lightbox-img');
      img.src = src;
      lightbox.classList.add('active');
      document.body.style.overflow = 'hidden';
    }

    function closeLightbox(e) {
      if (e.target.id === 'lightbox' || e.target.classList.contains('lightbox-close')) {
        document.getElementById('lightbox').classList.remove('active');
        document.body.style.overflow = '';
      }
    }

    document.addEventListener('keydown', function(e) {
      if (e.key === 'Escape') {
        document.getElementById('lightbox').classList.remove('active');
        document.body.style.overflow = '';
      }
    });
  </script>
